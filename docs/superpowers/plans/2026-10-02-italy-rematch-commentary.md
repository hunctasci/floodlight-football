**Goal:** Turn the silent v1 italy-rematch cut into a ~52 s comedy reel with an original English

faster-whisper, speechbrain ECAPA, TypeScript + Remotion, `node:test` via `tsx --test`.

## Review Focus

1. **Overlapping voices** — a long COMM take running into the next COMM/NINE line. Expect: no two lines overlap; the cut grows instead. Pinned in Task 3 (`no dialogue overlaps`).
2. **Missing assets at mix/edit time** — a cue name, a line id, or a take WAV referenced by `edit.json` that doesn't exist. Expect: a test fails before rendering. Pinned in Task 1 (cues + lines) and Task 3 (every placed line has a take).
3. **Silent Blender failures** — a shot raises but the batch reports success (seen on S14/S15). Expect: non-zero exit. Pinned in Task 6.
4. **Hero out of frame** — a camera that misses #9 (seen on S00_SH04, S04). Expect: the build fails with the offending time and screen coords. Pinned in Task 6 (guard) and used in Tasks 7–9.
5. **Score bug wrong at the edges** — bug visible on the end card, or flipping on the wrong shot. Expect: `TUR 1–4 ITA` from 0 s until the S13 wipe, `ITA 0–0 TUR · MON 5 OCT` after, hidden on S16. Pinned in Task 4.

```python
def score_motif(dur, seed=0):
    """The recurring 1–4 motif: one low BUM, then four dry ticks (the score as rhythm)."""
    n = int(SR * max(dur, 1.0))
    out = np.zeros(n)
    bum = sub_hit(0.6, seed)[:, 0]
    out[:len(bum)] += bum[: n] * 0.9
    tick = watch_click(0.08, seed)[:, 0]
    for k in range(4):
        i = int(SR * (0.32 + k * 0.13))
        out[i:i + len(tick)] += tick[: n - i] * 0.7
    return out


def elevator_ding(dur, seed=0):
    """Two-partial lift chime with a soft exponential ring-out."""
    t = np.arange(int(SR * max(dur, 0.9))) / SR
    env = np.exp(-t * 4.0)
    return (np.sin(2 * np.pi * 1318.5 * t) * 0.5 + np.sin(2 * np.pi * 1975.5 * t) * 0.25) * env * 0.6


def football_impact(dur, seed=0):
    """Genre switch: a sub drop under a hard ball slap."""
    n = int(SR * max(dur, 0.8))
    out = np.zeros(n)
    hit = sub_hit(0.8, seed)[:, 0]
    out[:len(hit)] += hit[:n]
    land = ball_land(0.5, seed)[:, 0]
    out[:len(land)] += land[:n] * 1.2
    return out
```

- [ ] **Step 3: Write `tools/diaries/py/voices_qwen.py`**

```python
"""Commentator takes with Qwen3-TTS VoiceDesign (Apache-2.0), local + offline via mlx-audio.

    $HNC_QWEN_PY tools/diaries/py/voices_qwen.py <dialogue.json> <vo_dir> [L03 …] [--takes=6]

Each line is designed exactly as the owner auditioned "Qwen B": the speaker's fixed `design` text plus
the line's `delivery` note. Design can drift the timbre take to take, so identity is held by selection:
among takes whose transcript matches best, keep the one whose ECAPA speaker embedding is closest to
vo/refs/<SPEAKER>.wav (the approved audition take). Fictional voice; no real person is cloned.
"""
import json
import sys
from pathlib import Path

import mlx.core as mx
import numpy as np
import soundfile as sf
import torch

sys.path.insert(0, str(Path(__file__).parent))
from voices_chatterbox import normalise, resample, trim, wer  # noqa: E402

SR = 48000
MODEL = "mlx-community/Qwen3-TTS-12Hz-1.7B-VoiceDesign-bf16"


def main():
    spec_path, vo = Path(sys.argv[1]), Path(sys.argv[2])
    only = {a for a in sys.argv[3:] if not a.startswith("--")}
    n_takes = next((int(a.split("=")[1]) for a in sys.argv[3:] if a.startswith("--takes=")), 6)
    spec = json.loads(spec_path.read_text())
    from faster_whisper import WhisperModel
    from mlx_audio.tts.utils import load_model
    from speechbrain.inference.speaker import EncoderClassifier
    tts = load_model(model_path=MODEL)
    asr = WhisperModel("base.en", device="cpu", compute_type="int8")
    spk = EncoderClassifier.from_hparams(source="speechbrain/spkrec-ecapa-voxceleb",
                                         savedir=str(Path.home() / ".cache/hnc/spkrec-ecapa"))

    def embed(a, sr):
        x = torch.from_numpy(resample(a.astype(np.float32), sr, 16000)).unsqueeze(0)
        e = spk.encode_batch(x).squeeze()
        return e / e.norm()

    refs = {}
    report_path = vo / "voices.json"
    report = json.loads(report_path.read_text()) if report_path.exists() else {}
    for line in spec["lines"]:
        if only and line["id"] not in only:
            continue
        sp = spec["speakers"][line["speaker"]]
        if sp.get("engine") != "qwen":
            continue
        if line["speaker"] not in refs:
            r, rsr = sf.read(vo / "refs" / f"{line['speaker']}.wav", dtype="float32")
            refs[line["speaker"]] = embed(r if r.ndim == 1 else r.mean(1), rsr)
        text = line["text"]
        spoken = line.get("say", text)
        instruct = f"{sp['design']} Delivery for this line: {line.get('delivery', 'natural')}."
        takes = []
        for k in range(n_takes):
            mx.random.seed(2000 + k * 31 + int(line["id"][1:]))
            chunks = [np.array(r.audio, dtype=np.float32) for r in
                      tts.generate(text=spoken, instruct=instruct, lang_code=sp.get("lang", "en"),
                                   temperature=line.get("temperature", sp.get("temperature", 0.9)))]
            a = trim(resample(np.concatenate(chunks), tts.sample_rate, SR), SR)
            hyp = " ".join(s.text.strip() for s in asr.transcribe(resample(a, SR, 16000), beam_size=5, language="en")[0])
            sim = float(torch.dot(embed(a, SR), refs[line["speaker"]]))
            takes.append((round(wer(spoken, hyp), 2), -sim, k, a, hyp, sim))
        takes.sort(key=lambda t: (t[0], t[1]))
        err, _neg, k, a, hyp, sim = takes[0]
        a = normalise(a)
        sf.write(vo / f"{line['id']}.wav", a, SR, subtype="PCM_24")
        report[line["id"]] = {"speaker": line["speaker"], "text": text, "take": k, "seconds": round(len(a) / SR, 3),
                              "asr": hyp, "wer": err, "similarity": round(sim, 3), "engine": "qwen"}
        flag = "" if err == 0 and sim >= 0.6 else "  <-- CHECK"
        print(f"{line['id']} {line['speaker']:5s} {len(a) / SR:5.2f}s sim={sim:.2f} asr='{hyp}' wer={err:.2f}{flag}", flush=True)
        report_path.write_text(json.dumps(report, indent=2) + "\n")


if __name__ == "__main__":
    main()
```

import React from 'react';
import type { ScoreBugSpec, TimedShot } from './edit';
import { DIARY_TYPE } from './type';

/** Which broadcast score the bug shows at `frame` (episode clock), or null when hidden. */
export function scoreBugText(shots: TimedShot[], fps: number, bug: ScoreBugSpec, frame: number): string | null {
  const shot = shots.find((s) => frame >= s.from && frame < s.from + s.frames);
  if (!shot || bug.hideOn.includes(shot.id)) return null;
  const flip = shots.find((s) => s.id === bug.flip.shot);
